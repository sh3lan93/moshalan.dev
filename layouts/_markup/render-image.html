{{- /* Markdown images. Overrides PaperMod's hook to add width/height (so the page does not jump while images load) and async decoding. Dimensions come from the file in assets/ or the page bundle; remote images and SVGs are left without them. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $size := dict -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if and (eq .MediaType.MainType "image") (ne .MediaType.SubType "svg") -}} {{- $size = dict "width" (string .Width) "height" (string .Height) -}} {{- end -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $attributes := merge $size .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy" "decoding" "async") -}} {{- /**/ -}}
